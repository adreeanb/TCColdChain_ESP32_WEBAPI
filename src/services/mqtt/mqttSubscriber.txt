import mqtt from 'mqtt';
import { CreateTemperatureReadingService } from '../telemetry/CreateTemperatureReadingService.js';
import { ReadingSourceType } from '@prisma/client';

const BROKER_URL = process.env.MQTT_BROKER_URL || 'mqtt://broker.hivemq.com:1883';

// ⚠️ ATENÇÃO: Para testes, cole aqui o ID de uma ColdUnit que você criou no seu banco de dados
const COLD_UNIT_TEST_ID = "COLE_O_ID_DA_UNIDADE_FRIA_AQUI"; 

export function iniciarMQTTSubscriber() {
  const client = mqtt.connect(BROKER_URL, {
    clientId: `node_backend_subscriber_${Math.random().toString(16).substring(2, 8)}`,
  });

  // Instancia o Service que contém toda a regra de negócio (Salvar + Alertas)
  const createTemperatureReadingService = new CreateTemperatureReadingService();

  client.on('connect', () => {
    console.log('✅ Backend conectado ao Broker MQTT público!');

    client.subscribe(['tcc/adrian/temperatura', 'tcc/adrian/temperatura/historico'], (err) => {
      if (!err) {
        console.log('📡 Inscrito nos tópicos:');
        console.log('   - tcc/adrian/temperatura (REALTIME)');
        console.log('   - tcc/adrian/temperatura/historico (STORE & FORWARD)');
      } else {
        console.error('❌ Erro ao se inscrever nos tópicos:', err);
      }
    });
  });

  client.on('message', async (topic, message) => {
    const payload = message.toString();

    try {
      if (topic === 'tcc/adrian/temperatura') {
        // LEITURA AO VIVO
        const tempValue = parseFloat(payload);

        if (!isNaN(tempValue)) {
          // Chama o Service em vez de chamar o Prisma diretamente
          await createTemperatureReadingService.execute({
            coldUnitId: COLD_UNIT_TEST_ID,
            temperature: tempValue,
            sourceType: ReadingSourceType.REALTIME // Usa o Enum do Prisma
          });
          
          console.log(`[MQTT -> SERVICE] REALTIME: ${tempValue} °C`);
        }
      } 
      
      else if (topic === 'tcc/adrian/temperatura/historico') {
        // LEITURA STORE AND FORWARD
        const dadosJson = JSON.parse(payload);
        const tempValue = parseFloat(dadosJson.temperatura);

        if (!isNaN(tempValue)) {
          await createTemperatureReadingService.execute({
            coldUnitId: COLD_UNIT_TEST_ID,
            temperature: tempValue,
            sourceType: ReadingSourceType.HISTORICAL, // Enum do Prisma
            
            // Caso o ESP32 envie o timestamp no JSON (opcional, senão usa a data atual)
            // timestamp: new Date(dadosJson.timestamp) 
          });
          
          console.log(`[MQTT -> SERVICE] HISTORICAL (Forward): ${tempValue} °C`);
        }
      }
    } catch (error: any) {
      // Captura erros, incluindo se o Service reclamar que a Unidade Fria não existe
      console.error(`❌ Erro MQTT ao processar tópico ${topic}:`, error.message);
    }
  });

  client.on('error', (err) => {
    console.error('❌ Erro no cliente MQTT:', err);
  });
}
